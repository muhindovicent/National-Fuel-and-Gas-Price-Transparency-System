function gIcon(v){return L.divIcon({html:'<div style="background:'+(v?'#008751':'#D32F2F')+';width:28px;height:28px;border-radius:6px;display:flex;align-items:center;justify-content:center;color:#fff;font-size:11px;border:2px solid #fff;box-shadow:0 2px 8px rgba(0,0,0,0.2);"><i class="fas fa-gas-pump"></i></div>',className:'',iconSize:[28,28],iconAnchor:[14,14]});}
function updMM(){
  if(!mapI)return;
  mapM.forEach(m=>mapI.removeLayer(m));mapM=[];
  const reg=document.getElementById('mapRegionFilter')?.value||'all';
  let f=reg==='all'?st:st.filter(s=>s.region.toLowerCase()===reg);
  f.sort((a,b)=>a.name.localeCompare(b.name));
  f.forEach(s=>{
    const m=L.marker([s.lat,s.lng],{icon:gIcon(s.verified)}).addTo(mapI);
    m.bindPopup('<div style="min-width:160px;font-family:Inter,sans-serif;"><div style="font-weight:700;font-size:12px;margin-bottom:4px;">'+s.name+'</div><div style="font-size:10px;color:#666;margin-bottom:6px;"><i class="fas fa-map-pin"></i> '+s.district+', '+s.region+'</div>'+FUEL.map(f=>'<div style="display:flex;justify-content:space-between;font-size:11px;padding:2px 0;"><span>'+f+'</span><strong>UGX '+s.prices[f].toLocaleString()+'</strong></div>').join('')+'<div style="margin-top:6px;display:flex;justify-content:space-between;font-size:10px;"><span><i class="fas fa-star" style="color:#F5A623;"></i> '+s.rating+'</span><span>'+(s.verified?'<span style="color:#008751;">&#10003; Verified</span>':'<span style="color:#D32F2F;">&#9888; Unverified</span>')+'</span></div></div>');
    mapM.push(m);
  });
  rMSL(f.slice(0,6));
}
